cmake_minimum_required(VERSION 3.20)
project(contact_router VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
option(CONTACT_SANITIZERS "Enable address and undefined-behavior checks" OFF)
add_library(contact src/router.cpp)
target_include_directories(contact PUBLIC include)
target_compile_features(contact PUBLIC cxx_std_20)
if(CONTACT_SANITIZERS)
  target_compile_options(contact PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(contact PUBLIC -fsanitize=address,undefined)
endif()
add_executable(contact-route src/main.cpp)
target_link_libraries(contact-route PRIVATE contact)
enable_testing()
add_executable(contact-tests tests/test_router.cpp)
target_link_libraries(contact-tests PRIVATE contact)
add_test(NAME routing-and-reservations COMMAND contact-tests)
